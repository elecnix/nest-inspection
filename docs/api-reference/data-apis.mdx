# Data APIs

The Nest Learning Thermostat Data APIs provide comprehensive access to data storage, configuration management, user data handling, and system logging. This document details all data APIs with their functions, parameters, and usage examples.

## Configuration API

### Overview
The Configuration API manages system and user configuration settings with persistence and validation.

```c
// Configuration API structure
typedef struct {
    // Configuration management
    config_handle_t (*open_config)(const char* config_name, access_mode_t mode);
    int (*close_config)(config_handle_t handle);
    int (*save_config)(config_handle_t handle);
    int (*reload_config)(config_handle_t handle);
    
    // Configuration value access
    int (*get_config_value)(config_handle_t handle, const char* key, 
                          config_value_t *value);
    int (*set_config_value)(config_handle_t handle, const char* key, 
                          config_value_t *value);
    int (*delete_config_value)(config_handle_t handle, const char* key);
    
    // Configuration sections
    config_section_t (*get_config_section)(config_handle_t handle, 
                                          const char* section_name);
    int (*create_config_section)(config_handle_t handle, 
                                const char* section_name);
    int (*delete_config_section)(config_handle_t handle, 
                                const char* section_name);
    
    // Configuration validation
    validation_result_t (*validate_config)(config_handle_t handle);
    int (*set_validation_schema)(config_handle_t handle, 
                                validation_schema_t *schema);
    
    // Configuration backup and restore
    int (*backup_config)(config_handle_t handle, const char* backup_path);
    int (*restore_config)(config_handle_t handle, const char* backup_path);
} configuration_api_t;

// Configuration value types
typedef enum {
    CONFIG_TYPE_STRING,                    // String value
    CONFIG_TYPE_INTEGER,                   // Integer value
    CONFIG_TYPE_FLOAT,                     // Float value
    CONFIG_TYPE_BOOLEAN,                   // Boolean value
    CONFIG_TYPE_BINARY,                    // Binary data
    CONFIG_TYPE_ARRAY,                     // Array value
    CONFIG_TYPE_OBJECT                     // Object value
} config_value_type_t;

// Configuration value structure
typedef struct {
    config_value_type_t type;              // Value type
    union {
        char string_value[256];            // String value
        int64_t integer_value;             // Integer value
        double float_value;                // Float value
        bool boolean_value;                // Boolean value
        struct {
            void* data;                    // Binary data
            size_t size;                   // Data size
        } binary_value;
        struct {
            config_value_t* elements;      // Array elements
            size_t count;                  // Element count
        } array_value;
        struct {
            config_pair_t* pairs;          // Object pairs
            size_t count;                  // Pair count
        } object_value;
    } value;
} config_value_t;

// Thermostat configuration example
void configure_thermostat_settings(void) {
    configuration_api_t *config_api = get_configuration_api();
    
    // Open thermostat configuration
    config_handle_t handle = config_api->open_config("thermostat_settings", 
                                                   ACCESS_MODE_READ_WRITE);
    
    // Set temperature range
    config_value_t temp_range = {
        .type = CONFIG_TYPE_OBJECT,
        .value.object_value = {
            .pairs = (config_pair_t[]){
                {"min", {.type = CONFIG_TYPE_FLOAT, .value.float_value = 16.0}},
                {"max", {.type = CONFIG_TYPE_FLOAT, .value.float_value = 30.0}},
                {"unit", {.type = CONFIG_TYPE_STRING, .value.string_value = "celsius"}}
            },
            .count = 3
        }
    };
    config_api->set_config_value(handle, "temperature_range", &temp_range);
    
    // Set schedule configuration
    config_value_t schedule = {
        .type = CONFIG_TYPE_BOOLEAN,
        .value.boolean_value = true
    };
    config_api->set_config_value(handle, "schedule_enabled", &schedule);
    
    // Set update interval
    config_value_t update_interval = {
        .type = CONFIG_TYPE_INTEGER,
        .value.integer_value = 30
    };
    config_api->set_config_value(handle, "update_interval_seconds", &update_interval);
    
    // Validate configuration
    validation_result_t validation = config_api->validate_config(handle);
    if (validation.valid) {
        config_api->save_config(handle);
    }
    
    config_api->close_config(handle);
}
```

```c
// Configuration template API
typedef struct {

// Template example
void setup_heating_system_template(void) {
    configuration_template_api_t *template_api = get_configuration_template_api();
    
    // Load heating system template
    template_handle_t template = template_api->load_template("heating_system");
    
    // Customize template for specific setup
    customization_t customization = {
        .overrides = (config_override_t[]){
            {"system_type", {.type = CONFIG_TYPE_STRING, .value.string_value = "gas_furnace"}},
            {"max_heating_temp", {.type = CONFIG_TYPE_FLOAT, .value.float_value = 35.0}},
            {"heating_stages", {.type = CONFIG_TYPE_INTEGER, .value.integer_value = 2}}
        },
        .override_count = 3
    };
    
    template_api->customize_template(template, &customization);
    
    // Apply to current configuration
    configuration_api_t *config_api = get_configuration_api();
    config_handle_t config = config_api->open_config("thermostat_settings", 
                                                   ACCESS_MODE_READ_WRITE);
    
    template_api->apply_template(config, template);
    config_api->save_config(config);
    
    template_api->save_template(template, "custom_heating_system");
    
    config_api->close_config(config);
    template_api->save_template(template, NULL);
}
```

## User Data API

### Overview
The User Data API manages user-specific data including preferences, schedules, and learned behavior.

```c
// User data API structure
typedef struct {
    // Data storage management
    data_handle_t (*open_user_data)(const char* data_path, access_mode_t mode);
    int (*close_user_data)(data_handle_t handle);
    
    // Data read/write operations
    ssize_t (*read_user_data)(data_handle_t handle, void* buffer, size_t size);
    ssize_t (*write_user_data)(data_handle_t handle, const void* buffer, size_t size);
    int (*seek_user_data)(data_handle_t handle, off_t offset, seek_origin_t origin);
    
    // Data synchronization
    int (*sync_user_data)(data_handle_t handle);
    int (*backup_user_data)(data_handle_t handle, const char* backup_path);
    int (*restore_user_data)(data_handle_t handle, const char* backup_path);
    
    // Data encryption and security
    int (*encrypt_user_data)(data_handle_t handle, encryption_key_t *key);
    int (*decrypt_user_data)(data_handle_t handle, encryption_key_t *key);
    int (*set_data_permissions)(data_handle_t handle, permissions_t permissions);
    
    // Data information
    data_info_t (*get_user_data_info)(data_handle_t handle);
    data_stats_t (*get_user_data_stats)(data_handle_t handle);
} user_data_api_t;

// User data types
typedef enum {
    USER_DATA_PREFERENCES,                 // User preferences
    USER_DATA_SCHEDULE,                    // Schedule data
    USER_DATA_LEARNED_BEHAVIOR,            // Learned behavior data
    USER_DATA_ENERGY_HISTORY,              // Energy history
    USER_DATA_USAGE_STATISTICS,            // Usage statistics
    USER_DATA_CUSTOM                       // Custom user data
} user_data_type_t;

// User data info
typedef struct {
    user_data_type_t type;                 // Data type
    size_t size;                           // Data size
    timestamp_t created;                   // Creation timestamp
    timestamp_t modified;                  // Last modification timestamp
    access_count_t access_count;           // Access count
    checksum_t checksum;                   // Data checksum
    encryption_status_t encryption;        // Encryption status
} data_info_t;

// User preferences example
void save_user_preferences(user_preferences_t *preferences) {
    user_data_api_t *user_data_api = get_user_data_api();
    
    // Open user preferences data
    data_handle_t handle = user_data_api->open_user_data("/data/user/preferences", 
                                                       ACCESS_MODE_READ_WRITE);
    
    // Serialize preferences
    size_t serialized_size;
    void* serialized_data = serialize_user_preferences(preferences, &serialized_size);
    
    // Write preferences to storage
    ssize_t written = user_data_api->write_user_data(handle, serialized_data, 
                                                   serialized_size);
    
    if (written == serialized_size) {
        // Sync data to persistent storage
        user_data_api->sync_user_data(handle);
        
        // Encrypt sensitive data
        encryption_key_t key = get_user_encryption_key();
        user_data_api->encrypt_user_data(handle, &key);
    }
    
    // Cleanup
    free(serialized_data);
    user_data_api->close_user_data(handle);
}

user_preferences_t* load_user_preferences(void) {
    user_data_api_t *user_data_api = get_user_data_api();
    
    // Open user preferences data
    data_handle_t handle = user_data_api->open_user_data("/data/user/preferences", 
                                                       ACCESS_MODE_READ);
    
    // Get data info
    data_info_t info = user_data_api->get_user_data_info(handle);
    
    // Allocate buffer and read data
    void* buffer = malloc(info.size);
    ssize_t read = user_data_api->read_user_data(handle, buffer, info.size);
    
    user_preferences_t* preferences = NULL;
    
    if (read == info.size) {
        // Decrypt if needed
        if (info.encryption == ENCRYPTION_ENABLED) {
            encryption_key_t key = get_user_encryption_key();
            user_data_api->decrypt_user_data(handle, &key);
        }
        
        // Deserialize preferences
        preferences = deserialize_user_preferences(buffer, info.size);
    }
    
    // Cleanup
    free(buffer);
    user_data_api->close_user_data(handle);
    
    return preferences;
}
```

```c
    entry_id_t entry_id;                   // Entry ID
    time_range_t time_range;               // Time range
    temperature_setpoint_t setpoint;       // Temperature setpoint
    schedule_type_t type;                  // Schedule type
    priority_t priority;                   // Entry priority
    metadata_t metadata;                   // Entry metadata
    timestamp_t created;                   // Creation timestamp
    timestamp_t modified;                  // Last modification timestamp

// Schedule management example
void create_weekly_schedule(void) {
    schedule_data_api_t *schedule_api = get_schedule_data_api();
    
    // Open weekly schedule
    schedule_handle_t schedule = schedule_api->open_schedule("weekly_schedule", 
                                                          ACCESS_MODE_READ_WRITE);
    
    // Create weekday schedule entries
    for (int day = 0; day < 5; day++) {
        // Morning entry
        schedule_entry_t morning = {
            .entry_id = generate_entry_id(),
            .time_range = {
                .start = {.hour = 6, .minute = 0, .day_of_week = day},
                .end = {.hour = 8, .minute = 30, .day_of_week = day}
            },
            .setpoint = {.temperature = 20.0, .type = SETPOINT_TYPE_HEATING},
            .type = SCHEDULE_TYPE_REGULAR,
            .priority = PRIORITY_NORMAL
        };
        schedule_api->add_schedule_entry(schedule, &morning);
        
        // Evening entry
        schedule_entry_t evening = {
            .entry_id = generate_entry_id(),
            .time_range = {
                .start = {.hour = 17, .minute = 0, .day_of_week = day},
                .end = {.hour = 23, .minute = 0, .day_of_week = day}
            },
            .setpoint = {.temperature = 22.0, .type = SETPOINT_TYPE_HEATING},
            .type = SCHEDULE_TYPE_REGULAR,
            .priority = PRIORITY_NORMAL
        };
        schedule_api->add_schedule_entry(schedule, &evening);
    }
    
    // Sync schedule to persistent storage
    schedule_api->sync_schedule(schedule);
    schedule_api->close_schedule(schedule);
}
```

```c
    data_backup_recovery_api_t *backup_api = get_data_backup_recovery_api();
    
    // Configure daily backup
    backup_config_t config = {
        .type = BACKUP_TYPE_INCREMENTAL,
        .scope = DATA_SCOPE_ALL,
        .compression = {
            .enabled = true,
            .algorithm = COMPRESSION_ALGORITHM_GZIP,
            .level = COMPRESSION_LEVEL_BALANCED
        },
        .encryption = {
            .enabled = true,
            .algorithm = ENCRYPTION_ALGORITHM_AES256,
            .key_source = KEY_SOURCE_DERIVED
        },
        .storage = {
            .type = STORAGE_TYPE_LOCAL,
            .path = "/backup/thermostat",
            .max_storage_size = 1024 * 1024 * 1024  // 1GB
        },
        .retention = {
            .policy = RETENTION_POLICY_TIME_BASED,
            .max_age = 30 * 24 * 60 * 60,    // 30 days
            .max_count = 50
        }
    };
    
    // Create backup schedule
    backup_schedule_t schedule = {
        .backup_config = config,
        .schedule_type = SCHEDULE_TYPE_DAILY,
        .execution_time = {.hour = 2, .minute = 0}, // 2:00 AM
        .retry_policy = {
            .max_retries = 3,
            .retry_interval = 3600         // 1 hour
        },
        .notification_config = {
            .on_success = false,
            .on_failure = true,
            .notification_method = NOTIFICATION_EMAIL
        }
    };
    
    schedule_id_t schedule_id = backup_api->schedule_backup(&schedule);
    set_backup_schedule_id(schedule_id);
}
```
